An attempt to use the **analog signal from the KY-037's AO pin** and compare its `analogRead()` value against a software-controlled threshold.

==Verdict: Informative experiment; the software-controlled AO approach is viable for observing and processing the analog signal, but a simple threshold comparison is not reliable for detecting short sound events such as claps.==

The goal was to let the Arduino manually determine whether the detected sound had reached the desired threshold, instead of relying on the KY-037's built-in **DO comparator**.

The project used:
- OLED to display the current AO reading and threshold
- Two buttons to increase/decrease the software threshold
- KY-037 AO for the analog signal
- Serial Monitor for observing readings

During testing, clapping produced only small and very rapid changes in the AO reading, while the KY-037's DO indicator could trigger much more easily. This demonstrated that a simple `analogRead()` → `if (value >= threshold)` approach can miss short sound transients because the Arduino is sampling the analog signal rather than continuously detecting threshold crossings like the module's comparator.


==main.cpp==
```c++
#include <Arduino.h>
#include <buttons.h>
#include <U8g2lib.h>
#include <conversion.h>


int btn1 = 7;
int btn2 = 6;
int sound_sensor = A0;
int sound_sensor_do = 5;
int led = 4;

//global var for functions to use
int btn1Reading;
int btn2Reading;

bool btn1State = false;
bool btn2State = false;
unsigned int threshold = 500;
  
//oled object creation
U8G2_SH1106_128X64_NONAME_1_HW_I2C oled(U8G2_R0);

void setup(){
  Serial.begin(9600);
  oled.begin();
  pinMode(btn1, INPUT);
  pinMode(btn2, INPUT);
  pinMode(led, OUTPUT);
  pinMode(sound_sensor,INPUT);
  pinMode(sound_sensor_do, INPUT);
}

void loop(){
  //oled
  oled.firstPage();
  //readings
  btn1Reading = digitalRead(btn1);
  btn2Reading = digitalRead(btn2);
  int sound_sensor_threshold = digitalRead(sound_sensor_do);
  unsigned long soundValue = analogRead(sound_sensor);
  
  //the function for handling the buttons
  //located at include/button.h
  handleButtons();

 //LOGIC FOR SOUND_SENSOR
 //Serial.println(soundValue);
//  if(sound_sensor_threshold == HIGH){
//   Serial.println("DETECTED");
//   //Serial.println(sound_sensor_threshold);
//  }
 if(soundValue >= threshold){
  digitalWrite(led, HIGH);
 }else{
  digitalWrite(led, LOW);
 }

//logic for display data in the oled
do{
  oled.setFont(u8g2_font_5x8_tr);
  oled.drawStr(3, 11, "Current Reading: ");
  oled.drawStr(97, 12, int_to_string(soundValue));
  oled.drawStr(5, 30, "Threshold: ");
  oled.drawStr(97, 31, int_to_string(threshold));
}while (oled.nextPage());
}
```

==buttons.h==
```c++

#include<Arduino.h>
#ifndef BUTTONS_H
#define BUTTONS_H


//using 'extern' syntax to use the variables in the main.cpp
extern int btn1;
extern int btn2;
extern bool btn1State;
extern bool btn2State;
extern int btn1Reading;
extern int btn2Reading;
extern unsigned int threshold;

void handleButtons(){
    //btn1 for decrements
    if(btn1Reading == HIGH && !btn1State){
        btn1State = true;
        threshold = threshold - 1;
        //Serial.print("threshold decrease: ");
        //Serial.println(threshold);
    }
    //btn2 for increase
    if(btn2Reading == HIGH && !btn2State){
        btn2State = true;
        threshold = threshold + 1;
        //Serial.print("threshold increase: ");
        //Serial.println(threshold);
    }

    //handles unpressed btn 1 and 2
    if(btn1Reading == LOW){
        btn1State = false;
    }

    if(btn2Reading == LOW){
        btn2State = false;
    }
}
#endif
```
